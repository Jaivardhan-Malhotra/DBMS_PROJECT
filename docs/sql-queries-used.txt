IPL MANAGEMENT SYSTEM
SQL QUERIES USED BY THE PROJECT
================================

How to read this file
---------------------
The question mark (?) is a parameter supplied by the backend with mysql2.
Do not replace a ? by concatenating user input into SQL. Use prepared
parameters, as the controllers do.

The examples below are grouped by the operation that uses them. They cover
the database scripts and the active API controllers. The source file shown
after each operation is the authoritative implementation.


1. DATABASE SETUP QUERIES
=========================

Create the database:

CREATE DATABASE ipl_management;

Create application tables:

CREATE TABLE venues (...);
CREATE TABLE seasons (...);
CREATE TABLE teams (...);
CREATE TABLE players (...);
CREATE TABLE matches (...);
CREATE TABLE player_match (...);

Add the team-captain relationship:

ALTER TABLE teams
    ADD CONSTRAINT fk_team_captain
    FOREIGN KEY (captain_id) REFERENCES players(player_id)
    ON DELETE SET NULL ON UPDATE CASCADE;

Create the match-details view:

CREATE VIEW v_match_details AS
SELECT
    m.match_id, m.match_date, m.match_time, m.match_type, m.status,
    m.win_margin, m.win_type, s.season_year, v.venue_name,
    v.city AS venue_city, t1.team_name AS team1_name,
    t1.short_name AS team1_short, t2.team_name AS team2_name,
    t2.short_name AS team2_short, w.team_name AS winner_name,
    p.player_name AS player_of_match_name
FROM matches m
JOIN seasons s ON m.season_id = s.season_id
JOIN venues v ON m.venue_id = v.venue_id
JOIN teams t1 ON m.team1_id = t1.team_id
JOIN teams t2 ON m.team2_id = t2.team_id
LEFT JOIN teams w ON m.winner_id = w.team_id
LEFT JOIN players p ON m.player_of_match = p.player_id;

Used in:
database/schema.sql

3. PLAYER QUERIES
=================

List players with their teams:

SELECT p.player_id, p.player_name, p.dob, p.nationality, p.role,
       p.batting_style, p.bowling_style, p.team_id, p.jersey_no,
       p.matches_played, p.runs_scored, p.wickets_taken,
       p.base_price, p.is_active, t.team_name, t.short_name,
       t.jersey_color
FROM players p
LEFT JOIN teams t ON p.team_id = t.team_id
WHERE p.player_name LIKE ?
ORDER BY p.player_name;

Get one player:

SELECT p.*, t.team_name, t.short_name, t.jersey_color
FROM players p
LEFT JOIN teams t ON p.team_id = t.team_id
WHERE p.player_id = ?;

Check a player's team before adding or changing the player:

SELECT team_id
FROM teams
WHERE team_id = ?;

Add a new player from the API:

    (player_name, dob, nationality, role, batting_style,
     bowling_style, team_id, jersey_no, is_active)
VALUES
    (?, ?, ?, ?, ?, ?, ?, ?, ?);

Return the newly added player:

SELECT p.player_id, p.player_name, p.dob, p.nationality, p.role,
       p.batting_style, p.bowling_style, p.team_id, p.jersey_no,
       p.is_active, t.team_name, t.short_name, t.jersey_color

LEFT JOIN teams t ON p.team_id = t.team_id

6. PLAYER STATISTICS QUERIES
============================

Get player statistics:

SELECT p.player_id, p.player_name, t.team_name,
       COUNT(pm.id) AS records_count,
       COUNT(DISTINCT pm.match_id) AS matches_count,
       COALESCE(SUM(pm.runs_scored), 0) AS total_runs,
       COALESCE(SUM(pm.wickets_taken), 0) AS total_wickets,
       COALESCE(SUM(pm.balls_faced), 0) AS total_balls,
       p.matches_played, p.runs_scored, p.wickets_taken
FROM players p
LEFT JOIN teams t ON t.team_id = p.team_id
LEFT JOIN player_match pm ON pm.player_id = p.player_id
GROUP BY p.player_id, p.player_name, t.team_name,
         p.matches_played, p.runs_scored, p.wickets_taken;

List match records:

SELECT pm.*, p.player_name, t.team_name, m.match_date,
       t1.team_name AS team1_name, t2.team_name AS team2_name
FROM player_match pm
JOIN players p ON p.player_id = pm.player_id
JOIN teams t ON t.team_id = pm.team_id
JOIN matches m ON m.match_id = pm.match_id
JOIN teams t1 ON t1.team_id = m.team1_id
JOIN teams t2 ON t2.team_id = m.team2_id
ORDER BY m.match_date DESC, pm.id DESC;

Add a match record:

INSERT INTO player_match
    (match_id, player_id, team_id, runs_scored, balls_faced,
     wickets_taken, is_out)
VALUES
    (?, ?, ?, ?, ?, ?, ?);

Update a match record:

UPDATE player_match
SET match_id = ?, player_id = ?, team_id = ?,
    runs_scored = ?, balls_faced = ?, wickets_taken = ?,
    is_out = ?
WHERE id = ?;

Delete a match record:

WHERE id = ?;

7. SEASON, VENUE, AND STANDING QUERIES
======================================

List seasons:

SELECT season_id, season_year
FROM seasons
ORDER BY season_year DESC;

List venues:

SELECT venue_id, venue_name, city
FROM venues
ORDER BY venue_name;

Select the requested or latest season:

SELECT season_id, season_year
FROM seasons
WHERE season_id = ?;

SELECT season_id, season_year
FROM seasons
WHERE season_year = (SELECT MAX(season_year) FROM seasons);

Calculate team standings for a season:

SELECT t.team_id, t.team_name,
       (SELECT COUNT(*) FROM matches m
        WHERE m.season_id = ? AND
              (m.team1_id = t.team_id OR m.team2_id = t.team_id) AND
              m.status = 'Completed') AS played,
       (SELECT COUNT(*) FROM matches m
        WHERE m.season_id = ? AND m.winner_id = t.team_id) AS wins,
       (SELECT COUNT(*) FROM matches m
        WHERE m.season_id = ? AND
              (m.team1_id = t.team_id OR m.team2_id = t.team_id) AND
              m.status = 'No Result') AS no_results
FROM teams t
ORDER BY t.team_name;

8. TEAM DATA MIGRATION QUERIES
==============================

Check required tables:

SELECT table_name
FROM information_schema.tables
WHERE table_schema = DATABASE()
  AND table_name IN ('teams', 'players');

Check whether the captain column exists:

SELECT column_name
FROM information_schema.columns
WHERE table_schema = DATABASE()
  AND table_name = 'teams'
  AND column_name = 'captain_id';

Add the captain column if needed:

ALTER TABLE teams
ADD COLUMN captain_id INT NULL AFTER coach;

Read team colors:

SELECT team_id, jersey_color
FROM teams;

Update a team's normalized jersey color:

UPDATE teams
SET jersey_color = ?
WHERE team_id = ?;
